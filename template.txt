ll gcd(ll a, ll b){
    return __gcd(a, b);

ll lcm(ll a, ll b){
    return (a / gcd(a, b)) * b;
}

ll power(ll a, ll b){
    ll ans = 1;

    while(b){
        if(b & 1)
            ans *= a;

        a *= a;
        b >>= 1;
    }

    return ans;
}

ll modPower(ll a, ll b, ll mod = MOD){
    ll ans = 1;
    a %= mod;

    while(b){
        if(b & 1)
            ans = (ans * a) % mod;

        a = (a * a) % mod;
        b >>= 1;
    }

    return ans;
}

ll modInverse(ll a, ll mod = MOD){
    return modPower(a, mod - 2, mod);
}

bool isPrime(ll n){

    if(n < 2)
        return false;

    for(ll i = 2; i * i <= n; i++){

        if(n % i == 0)
            return false;
    }

    return true;
}
